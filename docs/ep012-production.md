```bash
git init build/ep012-source
git -C build/ep012-source remote add origin https://github.com/zhipijun1996/map-edit.git
git -C build/ep012-source fetch --filter=blob:none --depth=1 origin 06bae30acb9709a24559c8ada532618be8500cac
python -m video_rush produce --source build/ep012-source --spec episodes/ep012.json --output build/episodes/ep012 --local-only
```
